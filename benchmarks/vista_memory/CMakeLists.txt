cmake_minimum_required(VERSION 3.18)
# Set before project(): CUDA compiler initialization otherwise supplies a default.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 90)
endif()
project(vista_memory LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
find_package(ZLIB REQUIRED)
foreach(name transfer memory spmv)
  add_executable(${name} ${name}.cu)
  target_link_libraries(${name} PRIVATE CUDA::cudart)
endforeach()
target_link_libraries(spmv PRIVATE CUDA::cusparse ZLIB::ZLIB)
add_executable(test_matrix test_matrix.cpp)
target_link_libraries(test_matrix PRIVATE ZLIB::ZLIB)
enable_testing()
add_test(NAME matrix_reader COMMAND test_matrix)
